# One executable per test file; each registered with CTest.
function(nightjar_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE nightjar)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME ${name} COMMAND ${name})
endfunction()

nightjar_test(test_pgm)
nightjar_test(test_file_replay_source)
nightjar_test(test_telemetry)
nightjar_test(test_gate_steps)
nightjar_test(test_motion_gate)
nightjar_test(test_conflating_slot)
nightjar_test(test_frame_ops)
nightjar_test(test_best_frame_selector)
nightjar_test(test_rule_engine)
